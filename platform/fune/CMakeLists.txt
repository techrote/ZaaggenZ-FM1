set(ZG_FUNE_SDK "${PROJECT_SOURCE_DIR}/third_party/fune")
add_library(zaaggenz-fune MODULE zaaggenz_plugin.c)
target_include_directories(zaaggenz-fune PRIVATE "${ZG_FUNE_SDK}/sdk" "${ZG_FUNE_SDK}/src")
target_link_libraries(zaaggenz-fune PRIVATE zaaggenz)
set_target_properties(zaaggenz-fune PROPERTIES PREFIX "" OUTPUT_NAME "zaaggenz")
if(NOT MSVC)
  target_compile_options(zaaggenz-fune PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(test-fune-adapter "${PROJECT_SOURCE_DIR}/tests/test_fune_adapter.c" zaaggenz_plugin.c)
target_include_directories(test-fune-adapter PRIVATE "${ZG_FUNE_SDK}/sdk" "${ZG_FUNE_SDK}/src")
target_compile_definitions(test-fune-adapter PRIVATE ZG_FUNE_TEST=1)
target_link_libraries(test-fune-adapter PRIVATE zaaggenz)
if(MSVC)
  target_compile_options(test-fune-adapter PRIVATE /UNDEBUG)
else()
  target_compile_options(test-fune-adapter PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
endif()
add_test(NAME fune-adapter COMMAND test-fune-adapter)
install(TARGETS zaaggenz-fune LIBRARY DESTINATION fune RUNTIME DESTINATION fune)
