cmake_minimum_required(VERSION 3.16)
project(zaaggenz_actual_fune_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT FUNE_SOURCE)
  message(FATAL_ERROR "Pass -DFUNE_SOURCE=/path/to/pinned/Fune")
endif()

get_filename_component(ZG_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(KREISEL_BUILD_SDL OFF CACHE BOOL "" FORCE)
set(KREISEL_BUILD_UI OFF CACHE BOOL "" FORCE)
set(KREISEL_SAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${FUNE_SOURCE}" fune EXCLUDE_FROM_ALL)
add_library(zg_actual_plugin MODULE
  "${ZG_ROOT}/platform/fune/zaaggenz_plugin.c"
  "${ZG_ROOT}/src/dsp.c" "${ZG_ROOT}/src/source_bank.c"
  "${ZG_ROOT}/src/presets.c" "${ZG_ROOT}/src/engine.c")
target_include_directories(zg_actual_plugin PRIVATE "${ZG_ROOT}/include" "${FUNE_SOURCE}/sdk" "${FUNE_SOURCE}/src")
set_target_properties(zg_actual_plugin PROPERTIES PREFIX "" OUTPUT_NAME "zaaggenz")
add_executable(zg_actual_host_test "${ZG_ROOT}/platform/fune/test_actual_host.c")
target_link_libraries(zg_actual_host_test PRIVATE kreisel_plugin_load)
target_compile_definitions(zg_actual_host_test PRIVATE ZG_PLUGIN_PATH="$<TARGET_FILE:zg_actual_plugin>")
if(MSVC)
  target_compile_options(zg_actual_host_test PRIVATE /UNDEBUG)
else()
  target_compile_options(zg_actual_host_test PRIVATE -UNDEBUG)
endif()
add_dependencies(zg_actual_host_test zg_actual_plugin)
enable_testing()
add_test(NAME actual_fune_host COMMAND zg_actual_host_test)

if(ZG_RECIPE_HEADER_DIR)
  add_executable(zg_fune_export "${ZG_ROOT}/platform/fune/export_presets.c")
  target_include_directories(zg_fune_export PRIVATE "${ZG_RECIPE_HEADER_DIR}")
  target_compile_definitions(zg_fune_export PRIVATE ZG_PLUGIN_PATH="$<TARGET_FILE:zg_actual_plugin>")
  target_link_libraries(zg_fune_export PRIVATE kreisel_plugin_load)
  if(MSVC)
    target_compile_options(zg_fune_export PRIVATE /UNDEBUG)
  else()
    target_compile_options(zg_fune_export PRIVATE -UNDEBUG)
  endif()
  add_dependencies(zg_fune_export zg_actual_plugin)
endif()

if(ZG_SONG_HEADER_DIR)
  add_executable(zg_fune_export_songs "${ZG_ROOT}/platform/fune/export_songs.c")
  target_include_directories(zg_fune_export_songs PRIVATE "${ZG_SONG_HEADER_DIR}")
  target_compile_definitions(zg_fune_export_songs PRIVATE ZG_PLUGIN_PATH="$<TARGET_FILE:zg_actual_plugin>")
  target_link_libraries(zg_fune_export_songs PRIVATE kreisel_plugin_load)
  if(NOT MSVC)
    target_link_libraries(zg_fune_export_songs PRIVATE m)
  endif()
  if(MSVC)
    target_compile_options(zg_fune_export_songs PRIVATE /UNDEBUG)
  else()
    target_compile_options(zg_fune_export_songs PRIVATE -UNDEBUG)
  endif()
  add_dependencies(zg_fune_export_songs zg_actual_plugin)
endif()
