cmake_minimum_required(VERSION 3.16)
project(ZaaggenZFM1 VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(ZG_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
option(ZG_BUILD_FUNE_HOST "Build the Fune desktop plugin and adapter tests" ON)
set(ZG_SOURCE_BANK_FILE "" CACHE FILEPATH "Optional compiler-generated custom source bank")
add_library(zaaggenz STATIC src/dsp.c src/source_bank.c src/engine.c src/presets.c)
set_target_properties(zaaggenz PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(zaaggenz PUBLIC include)
if(ZG_SOURCE_BANK_FILE)
  get_filename_component(ZG_SOURCE_BANK_FILE "${ZG_SOURCE_BANK_FILE}" ABSOLUTE)
  target_compile_definitions(zaaggenz PRIVATE ZG_SOURCE_BANK_INCLUDE="${ZG_SOURCE_BANK_FILE}")
endif()
if(MSVC)
  target_compile_options(zaaggenz PRIVATE /W4)
else()
  target_compile_options(zaaggenz PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ZG_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(zaaggenz PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_executable(zaag-render app/render.c)
target_link_libraries(zaag-render PRIVATE zaaggenz)
if(NOT WIN32)
  target_link_libraries(zaag-render PRIVATE m)
endif()
if(WIN32)
  add_executable(ZaaggenZ-Audition WIN32 app/win_player.c)
  target_link_libraries(ZaaggenZ-Audition PRIVATE zaaggenz winmm comctl32)
endif()
enable_testing()
add_executable(test-engine tests/test_engine.c)
target_link_libraries(test-engine PRIVATE zaaggenz)
add_test(NAME engine COMMAND test-engine)
add_executable(test-dsp tests/dsp/test_dsp.c)
target_link_libraries(test-dsp PRIVATE zaaggenz)
add_test(NAME dsp COMMAND test-dsp)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_source_bank.c")
  add_executable(test-source-bank tests/test_source_bank.c)
  target_link_libraries(test-source-bank PRIVATE zaaggenz)
  add_test(NAME source-bank COMMAND test-source-bank)
endif()
add_executable(dsp-probe tests/dsp/dsp_probe.c)
target_link_libraries(dsp-probe PRIVATE zaaggenz)
add_executable(source-probe tests/source_probe.c)
target_link_libraries(source-probe PRIVATE zaaggenz)
add_executable(pitch-probe tests/dsp/engine_pitch_probe.c)
target_link_libraries(pitch-probe PRIVATE zaaggenz)
if(ZG_BUILD_FUNE_HOST AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/platform/fune/CMakeLists.txt")
  add_subdirectory(platform/fune)
endif()
# Assertions are executable qualification, including Release builds.
foreach(test_target test-engine test-dsp test-source-bank test-fune-adapter)
  if(TARGET ${test_target})
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test_target} PRIVATE -UNDEBUG)
    endif()
  endif()
endforeach()
install(TARGETS zaag-render RUNTIME DESTINATION .)
if(WIN32)
  install(TARGETS ZaaggenZ-Audition RUNTIME DESTINATION .)
endif()
install(DIRECTORY presets/ DESTINATION presets)
install(FILES README.md LICENSE DESTINATION .)
